// Não compila, não é API do framework e não deve ser usado para operar banco.
// Asserções independentes atuais vivem nos testes; snapshots/histórico não foram removidos.
if (action==='restore') {
    if (options.resetConfirmation!==`RESTORE ${options.operation} ${options.dataset}`) throw new Error('Restore destrutivo exige confirmação separada exata');
    await idle(trx,context);
    for (const [table,rows] of Object.entries(saved.snapshot) as [string,any[]][]) {
        if (!Object.keys(shapes).includes(table) || rows.length>1000) throw new Error('Snapshot não autorizado');
        await trx(table).withSchema(context.dataSchema ?? context.schema).delete();
        if (rows.length) await trx(table).withSchema(context.dataSchema ?? context.schema).insert(rows);
    }
    const restored = await inspect(trx,context,options.operation);
    if (hash(restored)!==saved.checksum) throw new Error('Restore divergente; rollback');
    return {restored:true,historyPreserved:true,checksum:saved.checksum};
}
const row = await BatchJobModel.findById(options.id!,{...context,db:trx});
if (!row || row.status!=='completed' || row.transactional_state?.dataset!==options.dataset || row.transactional_state.operation!==options.operation) throw new Error('Validação exige execução completed correspondente');
let expected = saved.snapshot;
if (options.operation==='EXP-06') {
    const names = await trx.raw('SELECT id, UPPER(BTRIM(nome)) AS normalized FROM jsonb_to_recordset(?::jsonb) AS s(id text,nome text)',[JSON.stringify(saved.snapshot.teste_socios)]);
    const normalized = new Map(names.rows.map((r:any)=>[r.id,r.normalized]));
    expected = {...expected,teste_socios:expected.teste_socios.map((r:any)=>({...r,nome_normalizado:normalized.get(r.id),processado:true}))};
} else if (options.operation==='EXP-08') expected={...expected,teste_destino:expected.teste_socios.map(({id,nome,documento}:any)=>({id,nome,documento}))};
else expected={...expected,teste_migracao:expected.teste_migracao.filter((r:any)=>r.uf!=='SC')};
const divergences=canonical(data)===canonical(expected)?0:1;
if (divergences) throw new Error('Validação independente encontrou divergência de conjuntos/valores/origem');
return {id:row.id,divergences,validated:true,snapshotChecksum:saved.checksum};
